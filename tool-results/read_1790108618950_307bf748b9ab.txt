    95→    95→function checkCompatibility(): CompatResult {
    96→    96→  const reasons: string[] = [];
    97→    97→  const w = window.innerWidth;
    98→    98→  const h = window.innerHeight;
    99→    99→  if (w < 1024) reasons.push("عرض صفحه کمتر از ۱۰۲۴ پیکسل است (نسخه پژوهشی صرفاً دسکتاپ است).");
   100→   100→  if (h < 600) reasons.push("ارتفاع صفحه کمتر از ۶۰۰ پیکسل است.");
   101→   101→  const coarseOnly = window.matchMedia("(pointer: coarse)").matches && !window.matchMedia("(pointer: fine)").matches;
   102→   102→  if (coarseOnly) reasons.push("دسترسی صفحه‌لمسی تشخیص داده شد؛ آزمون نیازمند صفحه‌کلید فیزیکی است.");
   103→   103→  if (typeof performance?.now !== "function") reasons.push("مرورگر از اندازه‌گیری دقیق زمان پشتیبانی نمی‌کند.");
   104→   104→  if (typeof window.requestAnimationFrame !== "function") reasons.push("مرورگر قدیمی است.");
   105→   105→  const ua = navigator.userAgent;
   106→   106→  const deviceType = /iPad|Tablet/.test(ua) ? "tablet" : /Mobi|Android.*Mobile|iPhone/.test(ua) ? "mobile" : "desktop";
   107→   107→
   108→   108→  return {
   109→   109→    ok: reasons.length === 0,
   110→   110→    reasons,
   111→   111→    deviceType,
   112→   112→    screenWidth: w,
   113→   113→    screenHeight: h,
   114→   114→    fullscreenCapable: typeof document.fullscreenEnabled === "undefined" ? false : document.fullscreenEnabled,
   115→   115→  };
   116→   116→}

   122→   122→export function IatApp() {
   123→   123→  const [phase, setPhase] = useState<Phase>("loading-env");
   124→   124→  const [compat, setCompat] = useState<CompatResult | null>(null);
   125→   125→  const [compatError, setCompatError] = useState<string | null>(null);
   126→   126→  const [session, setSession] = useState<SessionInfo | null>(null);
   127→   127→  const [blockIdx, setBlockIdx] = useState(0);
   128→   128→  const [preloadProgress, setPreloadProgress] = useState(0);
   129→   129→  const [preloadFailed, setPreloadFailed] = useState(false);
   130→   130→  const [creating, setCreating] = useState(false);
   131→   131→  const [uploading, setUploading] = useState(false);
   132→   132→  const [result, setResult] = useState<CompleteResponse["result"] | null>(null);
   133→   133→  const [demographics, setDemographics] = useState({ age: "", gender: "", education: "", province: "" });
   134→   134→  const [consented, setConsented] = useState(false);
   135→   135→  const [demoError, setDemoError] = useState<string | null>(null);
   136→   136→
   137→   137→  // Quality tracking (refs — zero re-render impact)
   138→   138→  const fullscreenExitRef = useRef(0);
   139→   139→  const fullscreenDeniedRef = useRef(false);
   140→   140→  const visibilityCountRef = useRef(0);
   141→   141→  const awayMsRef = useRef(0);
   142→   142→  const hiddenAtRef = useRef<number | null>(null);
   143→   143→  const finishedRef = useRef(false);
   144→   144→  const sessionRef = useRef<SessionInfo | null>(null);
   145→   145→  sessionRef.current = session;
   146→   146→  const trialsRef = useRef<ClientTrialResult[]>([]);
   147→   147→  const phaseRef = useRef<Phase>("loading-env");
   148→   148→  phaseRef.current = phase;
   149→   149→
   150→   150→  // --- Compatibility check (once) ------------------------------------------
   151→   151→  useEffect(() => {
   152→   152→    const c = checkCompatibility();
   153→   153→    setCompat(c);
   154→   154→    if (!c.ok) {
   155→   155→      setPhase("incompatible");
   156→   156→      return;
   157→   157→    }
   158→   158→    // Pending-sync recovery: an upload failed previously?
   159→   159→    try {
   160→   160→      const raw = localStorage.getItem(BUFFER_KEY);
   161→   161→      if (raw) {
   162→   162→        const buf = JSON.parse(raw) as { sessionId: string; trials: ClientTrialResult[]; quality: ClientQualityMetadata };
   163→   163→        if (buf?.sessionId && Array.isArray(buf.trials) && buf.trials.length > 0) {
   164→   164→          setPhase("pending-sync");
   165→   165→          return;
   166→   166→        }
   167→   167→      }
   168→   168→    } catch {
   169→   169→      localStorage.removeItem(BUFFER_KEY);
   170→   170→    }
   171→   171→    setPhase("intro");
   172→   172→  }, []);
   173→   173→
   174→   174→  // --- Global quality event tracking ----------------------------------------
   175→   175→  useEffect(() => {
   176→   176→    const onFsChange = () => {
   177→   177→      if (!document.fullscreenElement) {
   178→   178→        fullscreenExitRef.current += 1;
   179→   179→      }
   180→   180→    };
   181→   181→    const onVis = () => {
   182→   182→      if (document.hidden) {
   183→   183→        hiddenAtRef.current = Date.now();
   184→   184→        visibilityCountRef.current += 1;
   185→   185→      } else if (hiddenAtRef.current !== null) {
   186→   186→        awayMsRef.current += Date.now() - hiddenAtRef.current;
   187→   187→        hiddenAtRef.current = null;
   188→   188→      }
   189→   189→    };
   190→   190→    const onHide = () => {
   191→   191→      // Best-effort abandon marker while a test is in progress
   192→   192→      const s = sessionRef.current;
   193→   193→      const p = phaseRef.current;
   194→   194→      if (s && !finishedRef.current && (p === "running" || p === "block-intro")) {
   195→   195→        try {
   196→   196→          navigator.sendBeacon(`/api/session/${s.sessionId}/abandon`);
   197→   197→        } catch {
   198→   198→          /* best effort */
   199→   199→        }
   200→   200→      }
   201→   201→    };
   202→   202→    document.addEventListener("fullscreenchange", onFsChange);
   203→   203→    document.addEventListener("visibilitychange", onVis);
   204→   204→    window.addEventListener("pagehide", onHide);
   205→   205→    return () => {
   206→   206→      document.removeEventListener("fullscreenchange", onFsChange);
   207→   207→      document.removeEventListener("visibilitychange", onVis);
   208→   208→      window.removeEventListener("pagehide", onHide);
   209→   209→    };
   210→   210→  }, []);
   211→   211→
   212→   212→  const qualityMeta = useCallback((): ClientQualityMetadata => {
   213→   213→    return {
   214→   214→      fullscreenExitCount: fullscreenExitRef.current,
   215→   215→      fullscreenDenied: fullscreenDeniedRef.current,
   216→   216→      visibilityChangeCount: visibilityCountRef.current,
   217→   217→      awayDurationMs: awayMsRef.current,
   218→   218→      screenWidth: compat?.screenWidth ?? 0,
   219→   219→      screenHeight: compat?.screenHeight ?? 0,
   220→   220→      deviceType: compat?.deviceType ?? "desktop",
   221→   221→      language: navigator.language ?? "fa",
   222→   222→    };
   223→   223→  }, [compat]);
   224→   224→
   225→   225→  // --- Flow actions ----------------------------------------------------------
   226→   226→
   227→   227→  const startCreation = useCallback(async () => {
   228→   228→    setCreating(true);
   229→   229→    setDemoError(null);
   230→   230→    try {
   231→   231→      const res = await fetch("/api/session", {
   232→   232→        method: "POST",
   233→   233→        headers: { "Content-Type": "application/json" },
   234→   234→        body: JSON.stringify({
   235→   235→          consent: true,
   236→   236→          age: parseInt(demographics.age, 10),
   237→   237→          gender: demographics.gender,
   238→   238→          education: demographics.education,
   239→   239→          province: demographics.province || null,
   240→   240→          environment: {
   241→   241→            screenWidth: compat?.screenWidth ?? 0,
   242→   242→            screenHeight: compat?.screenHeight ?? 0,
   243→   243→            deviceType: compat?.deviceType ?? "desktop",
   244→   244→            language: navigator.language,
   245→   245→          },
   246→   246→        }),
   247→   247→      });
   248→   248→      if (!res.ok) {
   249→   249→        const data = await res.json().catch(() => ({}));
   250→   250→        throw new Error(data.error ?? "خطا در ایجاد جلسه");
   251→   251→      }
   252→   252→      const data = (await res.json()) as SessionInfo;
   253→   253→      setSession(data);
   254→   254→      setPhase("preload");
   255→   255→    } catch (e) {
   256→   256→      setDemoError(e instanceof Error ? e.message : "خطای شبکه");
   257→   257→      setPhase("error");
   258→   258→    } finally {
   259→   259→      setCreating(false);
   260→   260→    }
   261→   261→  }, [demographics, compat]);
   262→   262→
   263→   263→  // --- Preload ---------------------------------------------------------------
   264→   264→  const runPreload = useCallback(async () => {
   265→   265→    const s = sessionRef.current;
   266→   266→    if (!s) return;
   267→   267→    setPreloadFailed(false);
   268→   268→    setPreloadProgress(0);
   269→   269→    const paths = [...new Set(s.plan.trials.filter((t) => t.stimulusType === "image").map((t) => t.stimulusPath))];
   270→   270→    let loaded = 0;
   271→   271→    let failures = 0;
   272→   272→    await Promise.all(
   273→   273→      paths.map(
   274→   274→        (src) =>
   275→   275→          new Promise<void>((resolve) => {
   276→   276→            const img = new Image();
   277→   277→            const done = (ok: boolean) => {
   278→   278→              if (ok) loaded += 1;
   279→   279→              else failures += 1;
   280→   280→              setPreloadProgress(loaded / Math.max(1, paths.length));
   281→   281→              resolve();
   282→   282→            };
   283→   283→            img.onload = () => {
   284→   284→              // ensure full decode before the first critical trial
   285→   285→              if (typeof img.decode === "function") {
   286→   286→                img.decode().then(() => done(true)).catch(() => done(true));
   287→   287→              } else done(true);
   288→   288→            };
   289→   289→            img.onerror = () => done(false);
   290→   290→            img.src = src;
   291→   291→          }),
   292→   292→      ),
   293→   293→    );
   294→   294→    if (failures > 0) {
   295→   295→      console.error(`[IAT] preload failed for ${failures}/${paths.length} images`);
   296→   296→      setPreloadFailed(true);
   297→   297→      return;
   298→   298→    }
   299→   299→    setPhase("ready-to-start");
   300→   300→  }, []);
   301→   301→
   302→   302→  // Auto-run preload when entering the preload phase (hook order is stable —
   303→   303→  // this effect lives above all conditional rendering).
   304→   304→  useEffect(() => {
   305→   305→    if (phase === "preload" && !preloadFailed && sessionRef.current) {
   306→   306→      void runPreload();
   307→   307→    }
   308→   308→  }, [phase, preloadFailed, runPreload]);
   309→   309→
   310→   310→  const beginTest = useCallback(async () => {
   311→   311→    const s = sessionRef.current;
   312→   312→    if (!s) return;
   313→   313→    // Fullscreen request must happen inside the user gesture
   314→   314→    try {
   315→   315→      if (document.fullscreenEnabled) {
   316→   316→        await document.documentElement.requestFullscreen();
   317→   317→      } else {
   318→   318→        fullscreenDeniedRef.current = true;
   319→   319→      }
   320→   320→    } catch {
   321→   321→      fullscreenDeniedRef.current = true;
   322→   322→    }
   323→   323→    // mark RUNNING (no further requests until completion)
   324→   324→    void fetch(`/api/session/${s.sessionId}/start`, { method: "POST" }).catch(() => undefined);
   325→   325→    setBlockIdx(0);
   326→   326→    setPhase("block-intro");
   327→   327→  }, []);
   328→   328→
   329→   329→  // --- Upload ------------------------------------------------------------------
   330→   330→  const uploadResults = useCallback(async () => {
   331→   331→    const s = sessionRef.current;
   332→   332→    if (!s) return;
   333→   333→    setUploading(true);
   334→   334→    setPhase("uploading");
   335→   335→    const payload = {
   336→   336→      trials: trialsRef.current,
   337→   337→      quality: qualityMeta(),
   338→   338→    };
   339→   339→    try {
   340→   340→      const res = await fetch(`/api/session/${s.sessionId}/complete`, {
   341→   341→        method: "POST",
   342→   342→        headers: { "Content-Type": "application/json" },
   343→   343→        body: JSON.stringify(payload),
   344→   344→      });
   345→   345→      if (!res.ok) {
   346→   346→        const data = await res.json().catch(() => ({}));
   347→   347→        if (res.status === 422) {
   348→   348→          // server rejected data — keep buffer for researcher review
   349→   349→          setPhase("sync-error");
   350→   350→          return;
   351→   351→        }
   352→   352→        throw new Error(data.error ?? "upload failed");
   353→   353→      }
   354→   354→      const data = (await res.json()) as CompleteResponse;
   355→   355→      finishedRef.current = true;
   356→   356→      localStorage.removeItem(BUFFER_KEY);
   357→   357→      if (document.fullscreenElement) document.exitFullscreen().catch(() => undefined);
   358→   358→      setResult(data.result);
   359→   359→      setPhase("result");
   360→   360→    } catch {
   361→   361→      // network failure → PENDING_SYNC (buffer already saved)
   362→   362→      setPhase("sync-error");
   363→   363→    } finally {
   364→   364→      setUploading(false);
   365→   365→    }
   366→   366→  }, [qualityMeta]);
   367→   367→
   368→   368→  // --- Block completion -------------------------------------------------------
   369→   369→  const onBlockDone = useCallback(
   370→   370→    (records: ClientTrialResult[]) => {
   371→   371→      const s = sessionRef.current;
   372→   372→      if (!s) return;
   373→   373→      trialsRef.current = trialsRef.current.concat(records);
   374→   374→      // buffer to localStorage (offline resilience)
   375→   375→      try {
   376→   376→        localStorage.setItem(
   377→   377→          BUFFER_KEY,
   378→   378→          JSON.stringify({ sessionId: s.sessionId, trials: trialsRef.current, quality: qualityMeta() }),
   379→   379→        );
   380→   380→      } catch {
   381→   381→        /* storage full — upload still attempted */
   382→   382→      }
   383→   383→      if (blockIdx + 1 < s.plan.blocks.length) {
   384→   384→        setBlockIdx(blockIdx + 1);
   385→   385→        setPhase("block-intro");
   386→   386→      } else {
   387→   387→        void uploadResults();
   388→   388→      }
   389→   389→    },
   390→   390→    [blockIdx, qualityMeta, uploadResults],
   391→   391→  );
   392→   392→
   393→   393→  const retryUpload = useCallback(async () => {
   394→   394→    // used both from sync-error and pending-sync recovery on reload
   395→   395→    try {
   396→   396→      const raw = localStorage.getItem(BUFFER_KEY);
   397→   397→      if (!raw) {
   398→   398→        setPhase("intro");
   399→   399→        return;
   400→   400→      }
   401→   401→      const buf = JSON.parse(raw) as { sessionId: string; trials: ClientTrialResult[]; quality: ClientQualityMetadata };
   402→   402→      setUploading(true);
   403→   403→      const res = await fetch(`/api/session/${buf.sessionId}/complete`, {
   404→   404→        method: "POST",
   405→   405→        headers: { "Content-Type": "application/json" },
   406→   406→        body: JSON.stringify({ trials: buf.trials, quality: buf.quality }),
   407→   407→      });
   408→   408→      const data = await res.json().catch(() => ({}));
   409→   409→      if (res.ok) {
   410→   410→        finishedRef.current = true;
   411→   411→        localStorage.removeItem(BUFFER_KEY);
   412→   412→        setResult((data as CompleteResponse).result);
   413→   413→        setPhase("result");
   414→   414→        return;
   415→   415→      }
   416→   416→      if (res.status === 422 || res.status === 404) {
   417→   417→        localStorage.removeItem(BUFFER_KEY);
   418→   418→        setPhase("intro");
   419→   419→        return;
   420→   420→      }
   421→   421→      setPhase("pending-sync");
   422→   422→    } catch {
   423→   423→      setPhase("pending-sync");
   424→   424→    } finally {
   425→   425→      setUploading(false);
   426→   426→    }
   427→   427→  }, []);
   428→   428→
   429→   429→  const discardBuffer = useCallback(() => {
   430→   430→    localStorage.removeItem(BUFFER_KEY);
   431→   431→    trialsRef.current = [];
   432→   432→    setPhase("intro");
   433→   433→  }, []);
   434→   434→
   435→   435→  const submitDemographics = useCallback(() => {
   436→   436→    const age = parseInt(demographics.age, 10);
   437→   437→    if (!Number.isFinite(age) || age < 8 || age > 100) {
   438→   438→      setDemoError("سن معتبر وارد کنید (۸ تا ۱۰۰).");
   439→   439→      return;
   440→   440→    }
   441→   441→    if (!demographics.gender) {
   442→   442→      setDemoError("جنسیت را انتخاب کنید.");
   443→   443→      return;
   444→   444→    }
   445→   445→    if (!demographics.education) {
   446→   446→      setDemoError("تحصیلات را انتخاب کنید.");
   447→   447→      return;
   448→   448→    }
   449→   449→    setDemoError(null);
   450→   450→    void startCreation();
   451→   451→  }, [demographics, startCreation]);
   452→   452→
   453→   453→  // --- Render ------------------------------------------------------------------
   454→   454→
   455→   455→  if (phase === "loading-env") {
   456→   456→    return <Centered>در حال بررسی محیط…</Centered>;
   457→   457→  }
   458→   458→
   459→   459→  if (phase === "incompatible" && compat) {
   460→   460→    return (
   461→   461→      <Centered>
