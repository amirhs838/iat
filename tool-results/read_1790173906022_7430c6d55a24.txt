    95→function checkCompatibility(): CompatResult {
    96→  // HARD requirements only: browser timing/rendering capabilities.
    97→  // Mobile phones and tablets are explicitly SUPPORTED (on-screen response
    98→  // buttons are the primary input; no keyboard is required). Screen size and
    99→  // pointer type are recorded as metadata, never used to block participation.
   100→  const reasons: string[] = [];
   101→  const w = window.innerWidth;
   102→  const h = window.innerHeight;
   103→  if (typeof performance?.now !== "function") reasons.push("مرورگر از اندازه‌گیری دقیق زمان پشتیبانی نمی‌کند.");
   104→  if (typeof window.requestAnimationFrame !== "function") reasons.push("مرورگر قدیمی است؛ لطفاً مرورگر را به‌روز کنید.");
   105→  const ua = navigator.userAgent;
   106→  const deviceType = /iPad|Tablet/.test(ua) ? "tablet" : /Mobi|Android.*Mobile|iPhone/.test(ua) ? "mobile" : "desktop";
   107→
   108→  return {
   109→    ok: reasons.length === 0,
   110→    reasons,
   111→    deviceType,
   112→    screenWidth: w,
   113→    screenHeight: h,
   114→    fullscreenCapable: typeof document.fullscreenEnabled === "undefined" ? false : document.fullscreenEnabled,
   115→  };
   116→}

   183→        hiddenAtRef.current = Date.now();
   184→        visibilityCountRef.current += 1;
   185→      } else if (hiddenAtRef.current !== null) {
   186→        awayMsRef.current += Date.now() - hiddenAtRef.current;
   187→        hiddenAtRef.current = null;
   188→      }
   189→    };

   191→      // Best-effort abandon marker while a test is in progress
   192→      const s = sessionRef.current;
   193→      const p = phaseRef.current;

   196→          navigator.sendBeacon(`/api/session/${s.sessionId}/abandon`);

   198→          /* best effort */

   200→      }
   201→    };

   203→    document.addEventListener("visibilitychange", onVis);
   204→    window.addEventListener("pagehide", onHide);

   207→      document.removeEventListener("visibilitychange", onVis);
   208→      window.removeEventListener("pagehide", onHide);

   382→      }

   398→        setPhase("intro");
   399→        return;
   400→      }

   402→      setUploading(true);
   403→      const res = await fetch(`/api/session/${buf.sessionId}/complete`, {
   404→        method: "POST",
   405→        headers: { "Content-Type": "application/json" },

   423→      setPhase("pending-sync");

   426→    }

   500→  }
   501→
   502→  if (phase === "intro") {
   503→    return (
   504→      <Centered>
   505→        <Card className="max-w-2xl w-full text-right">
   506→          <CardContent className="p-8 space-y-4">
   507→            <h2 className="text-2xl font-bold">آزمون تداعی ضمنی (IAT)</h2>
   508→            <div className="space-y-3 text-sm leading-7 text-muted-foreground">

   567→
   568→  if (phase === "demographics") {
   569→    return (

   579→                  id="age"
   580→                  type="number"

   582→                  max={100}
   583→                  inputMode="numeric"

   589→              </div>
   590→              <div className="space-y-1.5">
   591→                <Label>جنسیت</Label>
   592→                <Select value={demographics.gender} onValueChange={(v) => setDemographics((d) => ({ ...d, gender: v }))}>

   594→                    <SelectValue placeholder="انتخاب کنید" />

   596→                  <SelectContent>
   597→                    {GENDERS.map((g) => (
   598→                      <SelectItem key={g} value={g}>
   599→                        {g}

   684→        <Card className="max-w-lg w-full text-right">

   686→            <h2 className="text-xl font-bold">همه‌چیز آماده است</h2>

   691→              <li>در صورت امکان، حالت تمام‌صفحه فعال می‌شود.</li>
   692→              <li>کد ناشناس شما: <span className="font-mono text-foreground" dir="ltr">{session.anonymousId}</span></li>
   693→            </ul>
   694→            <Button size="lg" onClick={() => void beginTest()}>
   695→              شروع آزمون
   696→            </Button>
   697→          </CardContent>
   698→        </Card>
   699→      </Centered>
   700→    );
   701→  }
   702→
   703→  if (phase === "block-intro" && session) {
   704→    const block = session.plan.blocks[blockIdx];
   705→    return (
   706→      <BlockIntro
   707→        block={block}
   708→        index={blockIdx}
   709→        total={session.plan.blocks.length}
   710→        onStart={() => setPhase("running")}
   711→      />
   712→    );
   713→  }
   714→
   715→  if (phase === "running" && session) {
   716→    const block = session.plan.blocks[blockIdx];
   717→    const trials = session.plan.trials.filter((t) => t.blockNumber === block.number);
   718→    return <TrialRunner block={block} trials={trials} onBlockDone={onBlockDone} />;
   719→  }
   720→
   721→  if (phase === "uploading" || (phase === "sync-error" && uploading)) {
   722→    return <Centered>در حال ثبت داده‌ها… لطفاً صفحه را نبندید.</Centered>;
   723→  }
   724→
   725→  if (phase === "sync-error") {
   726→    return (
   727→      <Centered>
   728→        <Card className="max-w-lg w-full text-right">
   729→          <CardContent className="p-8 space-y-4">
   730→            <h2 className="text-xl font-bold">ارسال داده‌ها ناموفق بود</h2>
   731→            <p className="text-sm text-muted-foreground">

   748→    return (
   749→      <Centered>
   750→        <Card className="max-w-lg w-full text-right">
   751→          <CardContent className="p-8 space-y-3">
   752→            <h2 className="text-xl font-bold text-destructive">خطا</h2>
   753→            <p className="text-sm text-muted-foreground">{compatError ?? demoError ?? "خطای نامشخص"}</p>
   754→            <Button variant="outline" onClick={() => setPhase("intro")}>
   755→              بازگشت به ابتدا
   756→            </Button>
   757→          </CardContent>
   758→        </Card>
   759→      </Centered>
   760→    );
   761→  }
   762→
   763→  if (phase === "result" && result) {
   764→    return (
   765→      <Centered>
   766→        <Card className="max-w-2xl w-full text-right">
   767→          <CardContent className="p-8 space-y-5">

   794→
   795→  return <Centered>…</Centered>;
   796→}

   803→  return (

   810→function BlockIntro({
   811→  block,
   812→  index,
   813→  total,
   814→  onStart,
   815→}: {
   816→  block: BlockPlanEntry;
   817→  index: number;
   818→  total: number;
   819→  onStart: () => void;
   820→}) {
   821→  useEffect(() => {
   822→    const handler = (e: KeyboardEvent) => {
   823→      if (e.code === "Space" && !e.repeat) {
   824→        e.preventDefault();
   825→        onStart();
   826→      }
   827→    };
   828→    window.addEventListener("keydown", handler);
   829→    return () => window.removeEventListener("keydown", handler);
   830→  }, [onStart]);
   831→
   832→  return (
   833→    <div className="fixed inset-0 z-50 bg-white experiment-mode flex flex-col items-center justify-center p-6" data-testid="block-intro">
   834→      <div className="max-w-2xl w-full space-y-6">
   835→        <div className="flex items-center justify-between">
   836→          <h2 className="text-2xl font-bold">{block.instructionTitle}</h2>
   837→          <span className="text-sm text-neutral-400">
   838→            بخش {toFa(index + 1)} از {toFa(total)}
   839→          </span>
   840→        </div>
   841→        <div className="flex justify-between gap-4" dir="ltr" aria-hidden>
   842→          <span className="px-5 py-3 rounded-lg border-2 border-neutral-800 font-semibold text-lg text-center">
   843→            {block.leftLabel}
   844→          </span>
   845→          <span className="px-5 py-3 rounded-lg border-2 border-neutral-800 font-semibold text-lg text-center">
   846→            {block.rightLabel}
   847→          </span>
   848→        </div>
   849→        <p className="text-base leading-8 text-neutral-700 whitespace-pre-line text-center">
   850→          {block.instructionBody}
   851→        </p>
   852→        <div className="text-center space-y-3">
   853→          <p className="text-sm text-neutral-500">
   854→            برای شروع، دکمهٔ زیر را بزنید (روی رایانه، کلید Space هم کار می‌کند).
   855→          </p>
   856→          <Button size="lg" className="w-full sm:w-auto" onClick={onStart}>
   857→            شروع بخش {toFa(index + 1)}
   858→          </Button>
   859→        </div>
   860→      </div>
   861→    </div>
   862→  );
   863→}
